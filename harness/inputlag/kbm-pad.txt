# Input-lag profile, keyboard + mouse, then controller (run.sh --inputlag harness/inputlag/kbm-pad.txt).
# Every press alternates direction so the player stays on the spot; random gaps keep the presses off the frame phase.
activate
sleep 1
mark kbm-move
repeat 20
key w 0.25
sleep 0.35-0.65
key s 0.25
sleep 0.35-0.65
end
mark kbm-aim
mmove -8000 0
sleep 0.5
mdown right
sleep 1
repeat 15
mmove 8000 0
sleep 0.45-0.7
mmove -8000 0
sleep 0.45-0.7
end
mup right
sleep 1
mark kbm-aim-button
repeat 12
mouse right 0.4
sleep 0.5-0.8
end
mark pad-bind
flag inputlag_pad=1
sleep 3
mark pad-move
repeat 20
stick l 0 -1 0.25
sleep 0.35-0.65
stick l 0 1 0.25
sleep 0.35-0.65
end
mark pad-aim
sset r -1 0
sleep 0.3
pdown lt
sleep 1
repeat 15
sset r 1 0
sleep 0.45-0.7
sset r -1 0
sleep 0.45-0.7
end
pup lt
sleep 0.5
mark pad-aim-trigger
repeat 12
pad lt 0.4
sleep 0.5-0.8
end
sset r 0 0
done
